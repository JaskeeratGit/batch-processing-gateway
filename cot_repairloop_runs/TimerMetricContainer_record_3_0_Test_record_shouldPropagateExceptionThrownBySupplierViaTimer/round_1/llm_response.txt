package com.apple.spark.util;

import io.micrometer.core.instrument.Tag;
import io.micrometer.core.instrument.simple.SimpleMeterRegistry;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import java.util.Collection;
import java.util.Collections;
import java.util.function.Supplier;

import static org.junit.jupiter.api.Assertions.*;

/**
 * JUnit 5 test for TimerMetricContainer#record(Supplier, String, Collection)
 *
 * This test ensures that an exception thrown by the Supplier propagates through the Timer.record call.
 * Uses a real SimpleMeterRegistry to avoid having to mock internal MeterRegistry config/state needed by Timer.builder().register(...)
 */
class TimerMetricContainer_record_3_0_Test_record_shouldPropagateExceptionThrownBySupplierViaTimer {

    private SimpleMeterRegistry meterRegistry;
    private TimerMetricContainer container;

    @BeforeEach
    void setUp() {
        meterRegistry = new SimpleMeterRegistry();
        container = new TimerMetricContainer(meterRegistry);
    }

    @Test
    void record_shouldPropagateExceptionThrownBySupplierViaTimer() {
        String metricName = "error.metric";
        Collection<Tag> tags = Collections.emptyList();

        Supplier<String> throwingSupplier = () -> {
            throw new IllegalStateException("boom");
        };

        IllegalStateException ex = assertThrows(IllegalStateException.class,
            () -> container.record(throwingSupplier, metricName, tags));
        assertEquals("boom", ex.getMessage());

        // Ensure a timer was created for the metric name (sanity check)
        assertNotNull(meterRegistry.find(metricName).timer());
    }
}
