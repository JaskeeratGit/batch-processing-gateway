public class TimerMetricContainer_record_1_0_Test {

    // Utility to return reasonable defaults for proxy method return types
    private static Object defaultReturnValue(Class<?> returnType) {
        if (returnType == void.class)
            return null;
        if (returnType == boolean.class)
            return false;
        if (returnType == byte.class)
            return (byte) 0;
        if (returnType == short.class)
            return (short) 0;
        if (returnType == int.class)
            return 0;
        if (returnType == long.class)
            return 0L;
        if (returnType == float.class)
            return 0f;
        if (returnType == double.class)
            return 0d;
        return null;
    }

    @Test
    public void record_shouldInvokeTimerAndRunRunnable_whenTagsProvided() throws Exception {
        // Arrange
        AtomicInteger timerRecordCount = new AtomicInteger(0);
        // Create Timer proxy that runs the Runnable and increments counter
        Timer timerProxy = (Timer) Proxy.newProxyInstance(Timer.class.getClassLoader(), new Class[] { Timer.class }, new InvocationHandler() {

            @Override
            public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
                String name = method.getName();
                if ("record".equals(name) && args != null && args.length == 1 && args[0] instanceof Runnable) {
                    Runnable r = (Runnable) args[0];
                    r.run();
                    timerRecordCount.incrementAndGet();
                    return null;
                }
                // handle toString/equals/hashCode gracefully
                if ("toString".equals(name))
                    return "TimerProxy";
                if ("hashCode".equals(name))
                    return System.identityHashCode(proxy);
                if ("equals".equals(name))
                    return proxy == args[0];
                return defaultReturnValue(method.getReturnType());
            }
        });
        // Create MeterRegistry proxy that returns the Timer proxy for any timer(...) call
        MeterRegistry registryProxy = (MeterRegistry) Proxy.newProxyInstance(MeterRegistry.class.getClassLoader(), new Class[] { MeterRegistry.class }, new InvocationHandler() {

            @Override
            public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
                if ("timer".equals(method.getName())) {
                    // return our timer proxy regardless of name/tags
                    return timerProxy;
                }
                // handle toString/equals/hashCode gracefully
                if ("toString".equals(method.getName()))
                    return "MeterRegistryProxy";
                if ("hashCode".equals(method.getName()))
                    return System.identityHashCode(proxy);
                if ("equals".equals(method.getName()))
                    return proxy == args[0];
                return defaultReturnValue(method.getReturnType());
            }
        });
        String metricName = "test.metric";
        Tag tag = (Tag) Proxy.newProxyInstance(Tag.class.getClassLoader(), new Class[] { Tag.class }, (proxy, method, args) -> {
            if ("getKey".equals(method.getName()))
                return "k";
            if ("getValue".equals(method.getName()))
                return "v";
            if ("toString".equals(method.getName()))
                return "TagProxy";
            return defaultReturnValue(method.getReturnType());
        });
        Collection<Tag> tags = Arrays.asList(tag);
        TimerMetricContainer container = new TimerMetricContainer(registryProxy);
        // a side-effect flag to assert runnable ran
        final boolean[] ran = { false };
        Runnable runnable = () -> ran[0] = true;
        // Act
        container.record(runnable, metricName, tags);
        // Assert
        assertTrue(ran[0], "The runnable should have been executed by Timer.record");
        assertEquals(1, timerRecordCount.get(), "Timer.record should have been invoked exactly once");
        // Use reflection to access private field 'timers' and assert it has been populated (cache behavior)
        Field timersField = TimerMetricContainer.class.getDeclaredField("timers");
        timersField.setAccessible(true);
        @SuppressWarnings("unchecked")
        ConcurrentHashMap<?, ?> timersMap = (ConcurrentHashMap<?, ?>) timersField.get(container);
        assertNotNull(timersMap, "Timers map should be present");
        assertTrue(timersMap.size() >= 0, "Timers map should be accessible (size >= 0)");
    }

    @Test
    public void record_shouldPropagateExceptionThrownByRunnable_andStillCallTimer() throws Exception {
        // Arrange
        AtomicInteger timerRecordCount = new AtomicInteger(0);
        // Timer proxy that runs the Runnable (which will throw) and increments counter
        Timer timerProxy = (Timer) Proxy.newProxyInstance(Timer.class.getClassLoader(), new Class[] { Timer.class }, new InvocationHandler() {

            @Override
            public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
                String name = method.getName();
                if ("record".equals(name) && args != null && args.length == 1 && args[0] instanceof Runnable) {
                    Runnable r = (Runnable) args[0];
                    // exception from runnable will propagate
                    r.run();
                    timerRecordCount.incrementAndGet();
                    return null;
                }
                if ("toString".equals(name))
                    return "TimerProxy";
                if ("hashCode".equals(name))
                    return System.identityHashCode(proxy);
                if ("equals".equals(name))
                    return proxy == args[0];
                return defaultReturnValue(method.getReturnType());
            }
        });
        // MeterRegistry proxy that returns our timer proxy for timer(...) calls (including null tags)
        MeterRegistry registryProxy = (MeterRegistry) Proxy.newProxyInstance(MeterRegistry.class.getClassLoader(), new Class[] { MeterRegistry.class }, new InvocationHandler() {

            @Override
            public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
                if ("timer".equals(method.getName())) {
                    return timerProxy;
                }
                if ("toString".equals(method.getName()))
                    return "MeterRegistryProxy";
                if ("hashCode".equals(method.getName()))
                    return System.identityHashCode(proxy);
                if ("equals".equals(method.getName()))
                    return proxy == args[0];
                return defaultReturnValue(method.getReturnType());
            }
        });
        String metricName = "test.metric.exception";
        TimerMetricContainer container = new TimerMetricContainer(registryProxy);
        // Runnable that throws
        Runnable throwing = () -> {
            throw new RuntimeException("boom");
        };
        // Act & Assert
        RuntimeException ex = assertThrows(RuntimeException.class, () -> container.record(throwing, metricName, Collections.emptyList()));
        assertEquals("boom", ex.getMessage());
        assertEquals(0, timerRecordCount.get(), "Timer.record increments counter only after runnable completes; exception prevents increment");
        // Also assert reflection access to private 'timers' field exists and is a ConcurrentHashMap
        Field timersField = TimerMetricContainer.class.getDeclaredField("timers");
        timersField.setAccessible(true);
        Object timersObj = timersField.get(container);
        assertNotNull(timersObj);
        assertTrue(timersObj instanceof ConcurrentHashMap, "timers field should be a ConcurrentHashMap");
    }
}
