@ExtendWith(MockitoExtension.class)
public class AdminRest_version_3_0_Test {

    private AdminRest adminRest;

    @BeforeEach
    public void setUp() {
        // Create a mock of AdminRest without invoking its constructor and configure
        // the version() method to call the real implementation.
        adminRest = mock(AdminRest.class, withSettings().defaultAnswer(RETURNS_DEFAULTS));
        // Ensure calling version() executes the real method body.
        when(adminRest.version()).thenCallRealMethod();
    }

    @Test
    public void testVersionWritesJsonSuccessfully() throws Exception {
        Response response = adminRest.version();
        assertNotNull(response);
        assertEquals(200, response.getStatus());
        Object entity = response.getEntity();
        assertNotNull(entity);
        assertTrue(entity instanceof RestStreamingOutput);
        RestStreamingOutput streaming = (RestStreamingOutput) entity;
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        // Should not throw
        streaming.write(baos);
        String output = baos.toString("UTF-8");
        assertNotNull(output);
        String trimmed = output.trim();
        // Expect some JSON output (starts with '{' for object or '[' for array)
        assertFalse(trimmed.isEmpty());
        assertTrue(trimmed.startsWith("{") || trimmed.startsWith("["), "Output is not JSON: " + trimmed);
    }

    @Test
    public void testVersionHandlesExceptionWhenOutputStreamThrows() throws Exception {
        try (MockedStatic<ExceptionUtils> mockedExceptionUtils = mockStatic(ExceptionUtils.class)) {
            mockedExceptionUtils.when(ExceptionUtils::meterException).thenAnswer(invocation -> null);
            Response response = adminRest.version();
            assertNotNull(response);
            Object entity = response.getEntity();
            assertTrue(entity instanceof RestStreamingOutput);
            RestStreamingOutput streaming = (RestStreamingOutput) entity;
            OutputStream throwing = new OutputStream() {

                @Override
                public void write(int b) throws IOException {
                    throw new IOException("boom");
                }
            };
            // The method catches Throwable internally, so calling write should not throw.
            assertDoesNotThrow(() -> streaming.write(throwing));
            // Verify that ExceptionUtils.meterException() was invoked when the write failed.
            mockedExceptionUtils.verify(ExceptionUtils::meterException);
        }
    }
}
