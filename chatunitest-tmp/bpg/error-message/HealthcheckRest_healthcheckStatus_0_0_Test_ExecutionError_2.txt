/**

@ExtendWith(MockitoExtension.class)
public class HealthcheckRest_healthcheckStatus_0_0_Test {

    private AppConfig appConfig;

    private MeterRegistry meterRegistry;

    private HealthcheckRest healthcheckRest;

    private CounterMetricContainer mockCounter;

    @BeforeEach
    public void setUp() throws Exception {
        appConfig = new AppConfig();
        // Use a mock MeterRegistry to avoid SimpleMeterRegistry default-name issues
        meterRegistry = mock(MeterRegistry.class);
        // construct object under test
        healthcheckRest = new HealthcheckRest(appConfig, meterRegistry);
        // create mock CounterMetricContainer and inject into private field 'requestCounters'
        mockCounter = mock(CounterMetricContainer.class);
        Field f = HealthcheckRest.class.getDeclaredField("requestCounters");
        f.setAccessible(true);
        f.set(healthcheckRest, mockCounter);
    }

    @Test
    public void testHealthcheckStatus_returnsOKAndIncrementsCounter() {
        // arrange
        User mockUser = mock(User.class);
        when(mockUser.getName()).thenReturn("alice");
        AtomicReference<String> lastMetricName = new AtomicReference<>();
        AtomicReference<Tag[]> lastTags = new AtomicReference<>();
        // capture arguments when increment is called
        doAnswer(invocation -> {
            lastMetricName.set(invocation.getArgument(0));
            // varargs captured as Tag[]
            lastTags.set(invocation.getArgument(1, Tag[].class));
            return null;
        }).when(mockCounter).increment(anyString(), any(Tag[].class));
        // act
        HealthcheckResponse response = healthcheckRest.healthcheckStatus(mockUser);
        // assert response
        assertNotNull(response);
        assertEquals("OK", response.getStatus());
        // assert metric increment called and tags correct
        verify(mockCounter, times(1)).increment(anyString(), any(Tag[].class));
        assertNotNull(lastMetricName.get(), "Metric name should be set");
        Tag[] tags = lastTags.get();
        assertNotNull(tags, "Tags should not be null");
        assertEquals(2, tags.length, "Expected exactly two tags");
        Optional<Tag> nameTag = Arrays.stream(tags).filter(t -> "name".equals(t.getKey())).findFirst();
        Optional<Tag> userTag = Arrays.stream(tags).filter(t -> "user".equals(t.getKey())).findFirst();
        assertTrue(nameTag.isPresent(), "name tag should be present");
        assertEquals("healthcheck", nameTag.get().getValue());
        assertTrue(userTag.isPresent(), "user tag should be present");
        assertEquals("alice", userTag.get().getValue());
    }

    @Test
    public void testHealthcheckStatus_withDifferentUser_incrementsWithCorrectUserTag() {
        // arrange
        User mockUser = mock(User.class);
        when(mockUser.getName()).thenReturn("bob");
        AtomicReference<String> lastMetricName = new AtomicReference<>();
        AtomicReference<Tag[]> lastTags = new AtomicReference<>();
        doAnswer(invocation -> {
            lastMetricName.set(invocation.getArgument(0));
            lastTags.set(invocation.getArgument(1, Tag[].class));
            return null;
        }).when(mockCounter).increment(anyString(), any(Tag[].class));
        // act
        HealthcheckResponse response = healthcheckRest.healthcheckStatus(mockUser);
        // assert response
        assertNotNull(response);
        assertEquals("OK", response.getStatus());
        // assert metric increment called and tags contain the new user
        verify(mockCounter, times(1)).increment(anyString(), any(Tag[].class));
        Tag[] tags = lastTags.get();
        assertNotNull(tags);
        Optional<Tag> userTag = Arrays.stream(tags).filter(t -> "user".equals(t.getKey())).findFirst();
        assertTrue(userTag.isPresent());
        assertEquals("bob", userTag.get().getValue());
    }
}
