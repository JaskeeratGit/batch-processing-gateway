public class HealthcheckRest_healthcheckStatus_0_0_Test {

    private AppConfig appConfig;

    private SimpleMeterRegistry meterRegistry;

    private HealthcheckRest healthcheckRest;

    private SpyCounterMetricContainer spyCounter;

    // A small spy to capture increment calls
    private static class SpyCounterMetricContainer extends CounterMetricContainer {

        private volatile String lastMetricName;

        private volatile List<Tag> lastTags;

        public SpyCounterMetricContainer(SimpleMeterRegistry registry) {
            super(registry);
        }

        @Override
        public void increment(String metricName, Tag... tags) {
            this.lastMetricName = metricName;
            this.lastTags = Arrays.asList(tags);
            super.increment(metricName, tags);
        }

        public String getLastMetricName() {
            return lastMetricName;
        }

        public List<Tag> getLastTags() {
            return lastTags;
        }
    }

    @BeforeEach
    public void setUp() throws Exception {
        appConfig = new AppConfig();
        meterRegistry = new SimpleMeterRegistry();
        // construct object under test
        healthcheckRest = new HealthcheckRest(appConfig, meterRegistry);
        // create spy and inject into private field 'requestCounters'
        spyCounter = new SpyCounterMetricContainer(meterRegistry);
        Field f = HealthcheckRest.class.getDeclaredField("requestCounters");
        f.setAccessible(true);
        f.set(healthcheckRest, spyCounter);
    }

    @Test
    public void testHealthcheckStatus_returnsOKAndIncrementsCounter() {
        User user = new User("alice");
        HealthcheckResponse resp = healthcheckRest.healthcheckStatus(user);
        assertNotNull(resp, "Response should not be null");
        assertEquals("OK", resp.getStatus(), "Healthcheck status should be OK");
        // verify counter increment invoked
        String metricName = spyCounter.getLastMetricName();
        List<Tag> tags = spyCounter.getLastTags();
        assertNotNull(metricName, "Metric name passed to increment should not be null");
        assertNotNull(tags, "Tags passed to increment should not be null");
        // verify tags include name=healthcheck and user=alice
        boolean hasNameHealthcheck = tags.stream().anyMatch(t -> "name".equals(t.getKey()) && "healthcheck".equals(t.getValue()));
        boolean hasUserAlice = tags.stream().anyMatch(t -> "user".equals(t.getKey()) && "alice".equals(t.getValue()));
        assertTrue(hasNameHealthcheck, "Tags should include name=healthcheck");
        assertTrue(hasUserAlice, "Tags should include user=alice");
    }

    @Test
    public void testHealthcheckStatus_withDifferentUser_incrementsWithCorrectUserTag() {
        User user = new User("bob");
        HealthcheckResponse resp = healthcheckRest.healthcheckStatus(user);
        assertNotNull(resp);
        assertEquals("OK", resp.getStatus());
        List<Tag> tags = spyCounter.getLastTags();
        assertNotNull(tags);
        boolean hasUserBob = tags.stream().anyMatch(t -> "user".equals(t.getKey()) && "bob".equals(t.getValue()));
        assertTrue(hasUserBob, "Tags should include user=bob");
    }
}
