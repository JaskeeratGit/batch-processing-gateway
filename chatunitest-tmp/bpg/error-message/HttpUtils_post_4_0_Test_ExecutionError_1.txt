public class HttpUtils_post_4_0_Test {

    // Simple POJO for JSON parsing tests

        // No-arg constructor is required by Jackson
        public Person() {
        }
    }

    @Test
    public void testPostReturnsString_whenResponseIsJsonString() throws Exception {
        String json = "\"hello world\"";
        InputStream stream = new ByteArrayInputStream(json.getBytes());
        try (MockedStatic<HttpUtils> mocked = Mockito.mockStatic(HttpUtils.class)) {
            // Stub the internal 4-arg post(...) that returns the raw response string
            mocked.when(() -> HttpUtils.post("http://example", stream, "X-Header", "val")).thenReturn(json);
            // Let the generic 5-arg post(...) call the real method (so it will call our stubbed 4-arg and then parse)
            mocked.when(() -> HttpUtils.post("http://example", stream, "X-Header", "val", String.class)).thenCallRealMethod();
            String result = HttpUtils.post("http://example", stream, "X-Header", "val", String.class);
            assertEquals("hello world", result);
            // Additionally, use reflection to call the private parseJson(String, Class) method directly
            Method parseJson = HttpUtils.class.getDeclaredMethod("parseJson", String.class, Class.class);
            parseJson.setAccessible(true);
            @SuppressWarnings("unchecked")
            String reflected = (String) parseJson.invoke(null, json, String.class);
            assertEquals("hello world", reflected);
        }
    }

    @Test
    public void testPostParsesPojo_whenResponseIsJsonObject() throws Exception {
        String json = "{\"name\":\"Alice\",\"age\":30}";
        InputStream stream = new ByteArrayInputStream(json.getBytes());

            assertEquals("Alice", person.name);
            assertEquals(30, person.age);
            // Also verify using the private parseJson via reflection gives the same result
            Method parseJson = HttpUtils.class.getDeclaredMethod("parseJson", String.class, Class.class);
            parseJson.setAccessible(true);
            @SuppressWarnings("unchecked")

            assertEquals("Alice", reflected.name);
            assertEquals(30, reflected.age);
        }
    }
