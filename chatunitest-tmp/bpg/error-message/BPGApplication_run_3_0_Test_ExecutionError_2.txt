  Fixed unit test:
  - Removed use of MockitoExtension to avoid plugin initialization issues in some environments.
  - Uses Mockito static API directly.
  - Uses reflection to invoke a private method if present on BPGApplication.

public class BPGApplication_run_3_0_Test {

    private Environment environment;

    private JerseyEnvironment jerseyEnvironment;

    private MutableServletContextHandler servletContextHandler;

    private HealthCheckRegistry healthCheckRegistry;

    private MetricRegistry metricRegistry;

    @BeforeEach
    public void setUp() {
        // Create plain Mockito mocks (no MockitoExtension)
        environment = mock(Environment.class);
        jerseyEnvironment = mock(JerseyEnvironment.class);
        servletContextHandler = mock(MutableServletContextHandler.class);
        healthCheckRegistry = mock(HealthCheckRegistry.class);
        metricRegistry = new MetricRegistry();
        // Wire up expected environment behaviour
        when(environment.jersey()).thenReturn(jerseyEnvironment);
        when(environment.getApplicationContext()).thenReturn(servletContextHandler);
        when(environment.metrics()).thenReturn(metricRegistry);
        when(environment.healthChecks()).thenReturn(healthCheckRegistry);
        // Allow registering anything without side effects
        doNothing().when(jerseyEnvironment).register(any());
    }

    @Test
    public void testRun_nullContextPath_usesDefaultAndRegistersResources() throws Exception {
        // Prepare AppConfig mock
        AppConfig configuration = mock(AppConfig.class);
        when(configuration.getApplicationContextPath()).thenReturn(null);
        when(configuration.getAllowedUsers()).thenReturn(Collections.emptyList());
        when(configuration.getBlockedUsers()).thenReturn(Collections.emptyList());
        when(configuration.getSparkClusters()).thenReturn(Collections.emptyList());
        // Create application instance with monitorApplication = false
        BPGApplication app = new BPGApplication(false);
        // Run the application
        app.run(configuration, environment);
        // Verify that context path is set to default
        verify(servletContextHandler, times(1)).setContextPath(Constants.DEFAULT_APPLICATION_CONTEXT_PATH);
        // Verify that several resources were registered (at least once each)
        // Since register is called multiple times for different resources, we verify register was invoked.
        verify(jerseyEnvironment, atLeast(1)).register(any());
        // Verify that the MetricRegistry instance was stored in SharedMetricRegistries
        MetricRegistry regFromEnv = environment.metrics();
        assertNotNull(regFromEnv);
        assertSame(metricRegistry, regFromEnv);
    }

    @Test
    public void testRun_customContextPath_monitorApplicationTrue_and_invokePrivateSendPeriodicMetrics() throws Exception {
        // Prepare AppConfig mock with custom context path
        final String customContextPath = "/custom";
        AppConfig configuration = mock(AppConfig.class);
        when(configuration.getApplicationContextPath()).thenReturn(customContextPath);
        when(configuration.getAllowedUsers()).thenReturn(Arrays.asList("alice"));
        when(configuration.getBlockedUsers()).thenReturn(Arrays.asList("badguy"));
        when(configuration.getSparkClusters()).thenReturn(Collections.emptyList());
        // Create application instance with monitorApplication = true
        BPGApplication app = new BPGApplication(true);
        // Run the application (should attempt to start ApplicationMonitor, but we only verify interactions)
        app.run(configuration, environment);
        // Verify that custom context path is set
        verify(servletContextHandler, times(1)).setContextPath(customContextPath);
        // Verify that register was called for Jersey resources and features
        verify(jerseyEnvironment, atLeast(1)).register(any());
        // Attempt to invoke private sendPeriodicMetrics via reflection if present
        // It may be private; we reflectively locate and invoke it to satisfy testing of that code path.
        try {
            Method sendPeriodic = null;
            for (Method m : BPGApplication.class.getDeclaredMethods()) {
                if (m.getName().equals("sendPeriodicMetrics")) {
                    sendPeriodic = m;
                    break;
                }
            }
            if (sendPeriodic != null) {
                sendPeriodic.setAccessible(true);
                MeterRegistry meterRegistry = BPGStatsdConfig.createMeterRegistry();
                // The private method signature in the original code: sendPeriodicMetrics(configuration, meterRegistry)
                sendPeriodic.invoke(app, configuration, meterRegistry);
            } else {
                // If method not present, at least ensure createMeterRegistry works
                MeterRegistry meterRegistry = BPGStatsdConfig.createMeterRegistry();
                assertNotNull(meterRegistry);
            }
        } catch (InvocationTargetException ite) {
            // If the underlying method throws, rethrow cause for test visibility
            Throwable cause = ite.getCause();
            if (cause != null) {
                throw new RuntimeException(cause);
            } else {
                throw ite;
            }
        }
    }
}
