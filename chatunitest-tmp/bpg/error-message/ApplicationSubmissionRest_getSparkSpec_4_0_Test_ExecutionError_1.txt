/**
 * Unit tests for ApplicationSubmissionRest#getSparkSpec and the private removeEnvFromSpec method.
 */
public class ApplicationSubmissionRest_getSparkSpec_4_0_Test {

    // Helper subclass to override getSparkApplicationResource to return a mocked SparkApplication.
    static class TestApplicationSubmissionRest extends ApplicationSubmissionRest {

        private final SparkApplication returned;

        TestApplicationSubmissionRest(AppConfig config, SimpleMeterRegistry meterRegistry, SparkApplication returned) {
            super(config, meterRegistry);
            this.returned = returned;
        }

        @Override
        protected SparkApplication getSparkApplicationResource(String submissionId) {
            return returned;
        }
    }

    @Test
    public void testGetSparkSpec_removesEnvFromDriverAndExecutor() {
        // Arrange
        AppConfig cfg = new AppConfig();
        SimpleMeterRegistry meterRegistry = new SimpleMeterRegistry();
        SparkApplication sparkApp = mock(SparkApplication.class);
        SparkApplicationSpec spec = mock(SparkApplicationSpec.class);
        DriverSpec driver = mock(DriverSpec.class);
        ExecutorSpec executor = mock(ExecutorSpec.class);
        // driver and executor have non-null env lists (EnvVar)
        List<EnvVar> env = Collections.singletonList(mock(EnvVar.class));
        when(driver.getEnv()).thenReturn(env);
        when(executor.getEnv()).thenReturn(env);
        when(spec.getDriver()).thenReturn(driver);
        when(spec.getExecutor()).thenReturn(executor);
        when(sparkApp.getSpec()).thenReturn(spec);
        ApplicationSubmissionRest rest = new TestApplicationSubmissionRest(cfg, meterRegistry, sparkApp);
        // Act
        SparkApplicationSpec returned = rest.getSparkSpec("submission-1", "none", new User("alice"));
        // Assert
        // The same spec instance returned by our mock should be returned
        assertSame(spec, returned);
        // verify that setEnv(null) was invoked on both driver and executor (env removed)
        verify(driver).setEnv(null);
        verify(executor).setEnv(null);
    }

    @Test
    public void testGetSparkSpec_handlesNullEnvs() {
        // Arrange
        AppConfig cfg = new AppConfig();
        SimpleMeterRegistry meterRegistry = new SimpleMeterRegistry();
        SparkApplication sparkApp = mock(SparkApplication.class);
        SparkApplicationSpec spec = mock(SparkApplicationSpec.class);
        DriverSpec driver = mock(DriverSpec.class);
        ExecutorSpec executor = mock(ExecutorSpec.class);
        // both envs are null -> removeEnvFromSpec should not call setEnv
        when(driver.getEnv()).thenReturn(null);
        when(executor.getEnv()).thenReturn(null);
        when(spec.getDriver()).thenReturn(driver);
        when(spec.getExecutor()).thenReturn(executor);
        when(sparkApp.getSpec()).thenReturn(spec);
        ApplicationSubmissionRest rest = new TestApplicationSubmissionRest(cfg, meterRegistry, sparkApp);
        // Act
        SparkApplicationSpec returned = rest.getSparkSpec("submission-2", "none", new User("bob"));
        // Assert
        assertSame(spec, returned);
        // setEnv should NOT be called because env was already null
        verify(driver, never()).setEnv(anyList());
        verify(executor, never()).setEnv(anyList());
    }

    @Test
    public void testRemoveEnvFromSpec_exceptionHandled_viaReflection() throws Exception {
        // Arrange
        AppConfig cfg = new AppConfig();
        SimpleMeterRegistry meterRegistry = new SimpleMeterRegistry();
        SparkApplication sparkApp = mock(SparkApplication.class);
        SparkApplicationSpec spec = mock(SparkApplicationSpec.class);
        DriverSpec driver = mock(DriverSpec.class);
        ExecutorSpec executor = mock(ExecutorSpec.class);
        // Make driver.getEnv() throw to exercise the catch block inside removeEnvFromSpec
        when(driver.getEnv()).thenThrow(new RuntimeException("boom"));
        // executor returns a non-null env to ensure code path would have attempted setEnv if not for exception
        when(executor.getEnv()).thenReturn(Collections.singletonList(mock(EnvVar.class)));
        when(spec.getDriver()).thenReturn(driver);
        when(spec.getExecutor()).thenReturn(executor);
        when(sparkApp.getSpec()).thenReturn(spec);
        ApplicationSubmissionRest rest = new TestApplicationSubmissionRest(cfg, meterRegistry, sparkApp);
        // Use reflection to invoke private removeEnvFromSpec
        Method m = ApplicationSubmissionRest.class.getDeclaredMethod("removeEnvFromSpec", SparkApplicationSpec.class);
        m.setAccessible(true);
        // Act: should not throw even though driver.getEnv() throws inside method
        Object ret = m.invoke(rest, spec);
        // Assert: returned object should be the same spec (method returns passed spec)
        assertSame(spec, ret);
        // Because exception occurred before executor.setEnv(null) is reached, verify executor.setEnv was not called
        verify(executor, never()).setEnv(null);
    }
}
