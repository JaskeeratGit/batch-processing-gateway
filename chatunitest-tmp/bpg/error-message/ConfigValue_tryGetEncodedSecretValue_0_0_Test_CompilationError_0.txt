/*
Unit tests for ConfigValue.tryGetEncodedSecretValue(String)
Covers branches in:
- null / empty / plaintext / localhost env (present/missing)
- k8s:secret: malformed, namespace resolution, secret not found, no data, key missing, success
- retry path when client creation throws exceptions
Note: Provides lightweight test helpers in com.apple.spark.core package to simulate KubernetesHelper,
DefaultKubernetesClient and Secret APIs used by the production code.
*/
public class ConfigValue_tryGetEncodedSecretValue_0_0_Test {

    @AfterEach
    void tearDown() throws Exception {
        // Clear any fake env vars we set
        clearEnv();
        // Reset KubernetesHelper state
        KubernetesHelper.reset();
    }

    // Helper to call private static methods on ConfigValue via reflection
    private static Object invokePrivate(String methodName, String arg) throws Exception {
        Method m = ConfigValue.class.getDeclaredMethod(methodName, String.class);
        m.setAccessible(true);
        return m.invoke(null, arg);
    }

    @Test
    void testTryGetEncodedSecretValue_nullEmptyPlaintext() {
        assertNull(ConfigValue.tryGetEncodedSecretValue(null));
        assertEquals("", ConfigValue.tryGetEncodedSecretValue(""));
        String input = "plaintext:my-plain-value";
        assertEquals("my-plain-value", ConfigValue.tryGetEncodedSecretValue(input));
    }

    @Test
    void testTryGetEncodedSecretValue_localhostEnv_missingAndPresent() throws Exception {
        String valueMissing = "localhost:env:NON_EXISTENT_TEST_ENV_VAR";
        // ensure var not present
        clearEnv();
        // call via public method -> should return original since env var missing
        assertEquals(valueMissing, ConfigValue.tryGetEncodedSecretValue(valueMissing));
        // call private helper directly (reflection) with env present
        String envName = "TEST_CONFIGVALUE_ENV";
        setEnv(envName, "env-value-123");
        String valuePresent = "localhost:env:" + envName;
        // public call
        assertEquals("env-value-123", ConfigValue.tryGetEncodedSecretValue(valuePresent));
        // private call via reflection too
        Object fromPrivate = invokePrivate("tryGetEncodedSecretValueFromEnvVar", valuePresent);
        assertEquals("env-value-123", fromPrivate);
    }

    @Test
    void testTryGetEncodedSecretValue_k8s_malformedNamespaceMissing() throws Exception {
        // malformed (not 3 parts)
        String malformed = "k8s:secret:onlytwo";
        assertEquals(malformed, ConfigValue.tryGetEncodedSecretValue(malformed));
        Object fromPrivate = invokePrivate("tryGetEncodedSecretValueFromK8sSecret", malformed);
        assertEquals(malformed, fromPrivate);
        // namespace empty and KubernetesHelper.tryGetServiceAccountNamespace returns null -> original value
        String value = "k8s:secret::secretName:secretKey";
        // set SA namespace null
        KubernetesHelper.serviceAccountNamespace = null;
        assertEquals(value, ConfigValue.tryGetEncodedSecretValue(value));
    }

    @Test
    void testTryGetEncodedSecretValue_k8s_namespaceResolvedButSecretMissing() {
        // namespace empty but service account namespace provided
        String value = "k8s:secret::missingSecret:someKey";
        KubernetesHelper.serviceAccountNamespace = "resolved-ns";
        // KubernetesHelper.getLocalK8sClient by default returns a client with no secrets configured -> secret not found
        assertEquals(value, ConfigValue.tryGetEncodedSecretValue(value));
    }

    @Test
    void testTryGetEncodedSecretValue_k8s_secretNoDataAndKeyMissing_andSuccess() {
        // prepare a secret with null data
        String ns = "myns";
        String secretName = "s1";
        String key = "k1";
        // secret exists but data == null
        Secret secretWithNullData = new Secret(null);
        DefaultKubernetesClient client1 = DefaultKubernetesClient.forSecret(ns, secretName, secretWithNullData);
        // secret exists but no such key
        Map<String, String> dataMap = new HashMap<>();
        dataMap.put("otherKey", Base64.getEncoder().encodeToString("value-other".getBytes(StandardCharsets.UTF_8)));
        Secret secretWithOtherKey = new Secret(dataMap);
        DefaultKubernetesClient client2 = DefaultKubernetesClient.forSecret(ns, secretName, secretWithOtherKey);
        // success secret: with requested key
        Map<String, String> successData = new HashMap<>();
        successData.put(key, Base64.getEncoder().encodeToString("decoded-value-hello".getBytes(StandardCharsets.UTF_8)));
        Secret successSecret = new Secret(successData);
        DefaultKubernetesClient client3 = DefaultKubernetesClient.forSecret(ns, secretName, successSecret);
        // arrange KubernetesHelper to return client1 then client2 then client3 for subsequent calls
        KubernetesHelper.enqueueClient(client1);
        KubernetesHelper.enqueueClient(client2);
        KubernetesHelper.enqueueClient(client3);
        String valuePrefix = "k8s:secret:" + ns + ":" + secretName + ":" + key;
        // first attempt -> secret exists but data null -> returns original value
        String res1 = ConfigValue.tryGetEncodedSecretValue(valuePrefix);
        assertEquals(valuePrefix, res1);
        // second attempt -> secret exists but key missing -> returns original
        String res2 = ConfigValue.tryGetEncodedSecretValue(valuePrefix);
        assertEquals(valuePrefix, res2);
        // third attempt -> success -> returns decoded value
        String res3 = ConfigValue.tryGetEncodedSecretValue(valuePrefix);
        assertEquals("decoded-value-hello", res3);
    }

    @Test
    void testTryGetEncodedSecretValue_k8s_clientThrowsThenSucceeds_retryPath() {
        String ns = "retry-ns";
        String secretName = "secretRetry";
        String key = "rk";
        Map<String, String> successData = new HashMap<>();
        successData.put(key, Base64.getEncoder().encodeToString("retry-decode".getBytes(StandardCharsets.UTF_8)));
        Secret successSecret = new Secret(successData);
        DefaultKubernetesClient successClient = DefaultKubernetesClient.forSecret(ns, secretName, successSecret);
        // First two calls to getLocalK8sClient will throw; third returns successClient
        // will throw once
        KubernetesHelper.enqueueThrow();
        // will throw twice
        KubernetesHelper.enqueueThrow();
        // then return success
        KubernetesHelper.enqueueClient(successClient);
        String value = "k8s:secret:" + ns + ":" + secretName + ":" + key;
        String result = ConfigValue.tryGetEncodedSecretValue(value);
        assertEquals("retry-decode", result);
    }

    // Utilities to manage environment variables in tests
    @SuppressWarnings("unchecked")
    private static void setEnv(String key, String value) throws Exception {
        Map<String, String> newenv = new HashMap<>();
        newenv.put(key, value);
        try {
            Class<?> processEnvironmentClass = Class.forName("java.lang.ProcessEnvironment");
            Field theEnvironmentField = processEnvironmentClass.getDeclaredField("theEnvironment");
            theEnvironmentField.setAccessible(true);
            Map<String, String> env = (Map<String, String>) theEnvironmentField.get(null);
            env.putAll(newenv);
            Field theCaseInsensitiveEnvironmentField = processEnvironmentClass.getDeclaredField("theCaseInsensitiveEnvironment");
            theCaseInsensitiveEnvironmentField.setAccessible(true);
            Map<String, String> cienv = (Map<String, String>) theCaseInsensitiveEnvironmentField.get(null);
            cienv.putAll(newenv);
        } catch (NoSuchFieldException e) {
            // fallback for other JVMs
            Map<String, String> env = System.getenv();
            Class<?> envClass = env.getClass();
            try {
                Field m = envClass.getDeclaredField("m");
                m.setAccessible(true);
                Map<String, String> map = (Map<String, String>) m.get(env);
                map.putAll(newenv);
            } catch (NoSuchFieldException ex) {
                // last resort: reflect into Collections.UnmodifiableMap
                Field[] fields = envClass.getDeclaredFields();
                for (Field field : fields) {
                    if (field.getType().getName().contains("java.util.Map")) {
                        field.setAccessible(true);
                        Object obj = field.get(env);
                        if (obj instanceof Map) {
                            Map<String, String> map = (Map<String, String>) obj;
                            map.putAll(newenv);
                            break;
                        }
                    }
                }
            }
        }
    }

    @SuppressWarnings("unchecked")
    private static void clearEnv() throws Exception {
        try {
            Class<?> processEnvironmentClass = Class.forName("java.lang.ProcessEnvironment");
            Field theEnvironmentField = processEnvironmentClass.getDeclaredField("theEnvironment");
            theEnvironmentField.setAccessible(true);
            Map<String, String> env = (Map<String, String>) theEnvironmentField.get(null);
            env.clear();
            Field theCaseInsensitiveEnvironmentField = processEnvironmentClass.getDeclaredField("theCaseInsensitiveEnvironment");
            theCaseInsensitiveEnvironmentField.setAccessible(true);
            Map<String, String> cienv = (Map<String, String>) theCaseInsensitiveEnvironmentField.get(null);
            cienv.clear();
        } catch (NoSuchFieldException e) {
            Map<String, String> env = System.getenv();
            Class<?> envClass = env.getClass();
            try {
                Field m = envClass.getDeclaredField("m");
                m.setAccessible(true);
                Map<String, String> map = (Map<String, String>) m.get(env);
                map.clear();
            } catch (NoSuchFieldException ex) {
                // best-effort: try to remove entries by reflection into underlying map
                Field[] fields = envClass.getDeclaredFields();
                for (Field field : fields) {
                    if (field.getType().getName().contains("java.util.Map")) {
                        field.setAccessible(true);
                        Object obj = field.get(env);
                        if (obj instanceof Map) {
                            Map<String, String> map = (Map<String, String>) obj;
                            map.clear();
                            break;
                        }
                    }
                }
            }
        }
    }
}

/*
 Lightweight test-support classes that mimic the minimal API used by ConfigValue to interact with Kubernetes.
 These are in the same package (com.apple.spark.core) so ConfigValue will reference them at runtime during tests.
*/
class KubernetesHelper {

    // controllable service account namespace used when namespace part is empty
    static String serviceAccountNamespace = null;

    // queue of clients to return or throw markers
    private static final Deque<Object> clientQueue = new ArrayDeque<>();

    static void reset() {
        serviceAccountNamespace = null;
        clientQueue.clear();
    }

    static void enqueueClient(DefaultKubernetesClient client) {
        clientQueue.addLast(client);
    }

    static void enqueueThrow() {
        clientQueue.addLast(new ThrowMarker());
    }

    // Methods expected by ConfigValue
    static String tryGetServiceAccountNamespace() {
        return serviceAccountNamespace;
    }

    static DefaultKubernetesClient getLocalK8sClient() {
        Object item = clientQueue.pollFirst();
        if (item == null) {
            // default client that has no secrets configured (so get() returns null)
            return DefaultKubernetesClient.emptyClient();
        }
        if (item instanceof ThrowMarker) {
            throw new RuntimeException("Simulated client creation failure");
        }
        return (DefaultKubernetesClient) item;
    }

    private static class ThrowMarker {
    }
}

class DefaultKubernetesClient implements AutoCloseable {

    // map namespace -> (name -> Secret)
    private final Map<String, Map<String, Secret>> store = new HashMap<>();

    private DefaultKubernetesClient() {
    }

    static DefaultKubernetesClient emptyClient() {
        return new DefaultKubernetesClient();
    }

    static DefaultKubernetesClient forSecret(String namespace, String name, Secret secret) {
        DefaultKubernetesClient c = new DefaultKubernetesClient();
        c.store.computeIfAbsent(namespace, k -> new HashMap<>()).put(name, secret);
        return c;
    }

    SecretsOperations secrets() {
        return new SecretsOperations(this);
    }

    Secret getSecret(String namespace, String name) {
        Map<String, Secret> nsmap = store.get(namespace);
        if (nsmap == null)
            return null;
        return nsmap.get(name);
    }

    @Override
    public void close() {
        // no-op for test
    }

    // fluent API mimic
    static class SecretsOperations {

        private final DefaultKubernetesClient client;

        SecretsOperations(DefaultKubernetesClient client) {
            this.client = client;
        }

        InNamespace inNamespace(String namespace) {
            return new InNamespace(client, namespace);
        }
    }

    static class InNamespace {

        private final DefaultKubernetesClient client;

        private final String namespace;

        InNamespace(DefaultKubernetesClient client, String namespace) {
            this.client = client;
            this.namespace = namespace;
        }

        WithName withName(String name) {
            return new WithName(client, namespace, name);
        }
    }

    static class WithName {

        private final DefaultKubernetesClient client;

        private final String namespace;

        private final String name;

        WithName(DefaultKubernetesClient client, String namespace, String name) {
            this.client = client;
            this.namespace = namespace;
            this.name = name;
        }

        Secret get() {
            return client.getSecret(namespace, name);
        }
    }
}

class Secret {

    // data map: key -> base64 string
    private final Map<String, String> data;

    Secret(Map<String, String> data) {
        this.data = data;
    }

    public Map<String, String> getData() {
        return data;
    }
}
