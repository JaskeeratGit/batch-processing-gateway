    @Test
    public void testDefaultConstructor_and_getName_notNull() throws Exception {
        // Use default constructor
        BPGApplication app = new BPGApplication();
        assertNotNull(app);
        // getName should be callable and return a non-null value
        Method getNameMethod = BPGApplication.class.getMethod("getName");
        Object name = getNameMethod.invoke(app);
        assertNotNull(name, "getName() should not return null");
        assertTrue(name instanceof String, "getName() should return a String");
        assertFalse(((String) name).isEmpty(), "getName() should not be empty");
    }

    @Test
    public void testBooleanConstructor_setsMonitorApplicationField_true() throws Exception {
        // Use constructor with boolean argument
        Constructor<BPGApplication> ctor = BPGApplication.class.getConstructor(boolean.class);
        BPGApplication app = ctor.newInstance(true);
        assertNotNull(app);
        // Access the private field 'monitorApplication' via reflection
        Field monitorField = BPGApplication.class.getDeclaredField("monitorApplication");
        monitorField.setAccessible(true);
        Object fieldValue = monitorField.get(app);
        assertTrue(fieldValue instanceof Boolean);
        assertTrue((Boolean) fieldValue, "monitorApplication should be true for constructor(true)");
    }

    @Test
    public void testRun_nullContextPath_usesDefaultAndRegistersResources() {
        AppConfig config = mock(AppConfig.class);
        when(config.getApplicationContextPath()).thenReturn(null);
        when(config.getAllowedUsers()).thenReturn(Collections.emptyList());
        when(config.getBlockedUsers()).thenReturn(Collections.emptyList());
        when(config.getSparkClusters()).thenReturn(Collections.emptyList());
        BPGApplication app = new BPGApplication(false);
        // Execute
        app.run(config, environment);
        // Verify default context path used
        verify(servletContextHandler).setContextPath(Constants.DEFAULT_APPLICATION_CONTEXT_PATH);
        // Verify that core REST resources are registered
        verify(jerseyEnvironment).register(isA(ApplicationSubmissionRest.class));
        verify(jerseyEnvironment).register(isA(ApplicationGetLogRest.class));
        verify(jerseyEnvironment).register(isA(CloudStorageRest.class));
        verify(jerseyEnvironment).register(isA(AdminRest.class));
        verify(jerseyEnvironment).register(isA(HealthcheckRest.class));
        // Verify exception mapper and auth bindings registered (at least once)
        verify(jerseyEnvironment, atLeastOnce()).register(isA(ThrowableExceptionMapper.class));
        verify(jerseyEnvironment, atLeastOnce()).register(isA(io.dropwizard.auth.AuthDynamicFeature.class));
        verify(jerseyEnvironment, atLeastOnce()).register(isA(io.dropwizard.auth.AuthValueFactoryProvider.Binder.class));
        // Verify health check registered
        verify(healthCheckRegistry).register(eq("sparkClusters"), isA(BPGHealthCheck.class));
        // Verify metric registry was added to SharedMetricRegistries
        assertSame(metricRegistry, SharedMetricRegistries.getOrCreate(Constants.DEFAULT_METRIC_REGISTRY));
    }
