/**
 * JUnit 5 tests for KubernetesHelper.closeQuietly(Closeable)
 */
public class KubernetesHelper_closeQuietly_7_0_Test {

    private VarHandle loggerHandle;

    private Logger originalLogger;

    private Logger testLoggerProxy;

    private TestLoggerHandler handler;

    @BeforeEach
    public void setUp() throws Throwable {
        // Obtain a VarHandle to the private static final logger field and replace it with a proxy logger
        MethodHandles.Lookup lookup = MethodHandles.privateLookupIn(KubernetesHelper.class, MethodHandles.lookup());
        loggerHandle = lookup.findStaticVarHandle(KubernetesHelper.class, "logger", Logger.class);
        originalLogger = (Logger) loggerHandle.get();
        handler = new TestLoggerHandler();
        testLoggerProxy = (Logger) Proxy.newProxyInstance(Logger.class.getClassLoader(), new Class[] { Logger.class }, handler);
        loggerHandle.set(testLoggerProxy);
    }

    @AfterEach
    public void tearDown() throws Throwable {
        // Restore original logger
        if (loggerHandle != null && originalLogger != null) {
            loggerHandle.set(originalLogger);
        }
    }

    @Test
    public void testCloseQuietly_nullArgument_logsAndNoThrow() {
        // Passing null should be caught by the method and result in a warning logged (no exception thrown)
        assertDoesNotThrow(() -> KubernetesHelper.closeQuietly(null));
        assertEquals(1, handler.getWarnCount(), "Expected one warn call when null is passed");
        assertNotNull(handler.getLastMessage());
        assertTrue(handler.getLastMessage().contains("Failed to close"));
        Throwable t = handler.getLastThrowable();
        assertNotNull(t);
        assertTrue(t instanceof NullPointerException);
    }

    @Test
    public void testCloseQuietly_logsWhenCloseThrowsIOException() {
        Closeable closeable = () -> {
            throw new IOException("simulated IO");
        };
        // Should catch and log the Throwable
        KubernetesHelper.closeQuietly(closeable);
        assertEquals(1, handler.getWarnCount(), "Expected one warn call");
        assertNotNull(handler.getLastMessage());
        assertTrue(handler.getLastMessage().contains("Failed to close"));
        Throwable t = handler.getLastThrowable();
        assertNotNull(t);
        assertTrue(t instanceof IOException);
        assertEquals("simulated IO", t.getMessage());
    }

    @Test
    public void testCloseQuietly_logsWhenCloseThrowsError() {
        Closeable closeable = () -> {
            throw new Error("simulated error");
        };
        // Should catch and log the Throwable (even Error)
        KubernetesHelper.closeQuietly(closeable);
        assertEquals(1, handler.getWarnCount(), "Expected one warn call");
        Throwable t = handler.getLastThrowable();
        assertNotNull(t);
        assertTrue(t instanceof Error);
        assertEquals("simulated error", t.getMessage());
    }

    @Test
    public void testCloseQuietly_closesSuccessfully_noWarning() {
        final boolean[] closed = { false };
        Closeable closeable = () -> closed[0] = true;
        KubernetesHelper.closeQuietly(closeable);
        assertTrue(closed[0], "Closeable should have been closed");
        assertEquals(0, handler.getWarnCount(), "No warnings should be logged when close succeeds");
    }

    // InvocationHandler that records warn(...) calls
    private static class TestLoggerHandler implements InvocationHandler {

        private int warnCount = 0;

        private String lastMessage = null;

        private Throwable lastThrowable = null;

        @Override
        public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
            String name = method.getName();
            if ("warn".equals(name)) {
                // try to extract message and throwable in common signatures
                if (args != null) {
                    // warn(String msg, Throwable t)
                    if (args.length == 2 && args[0] instanceof String && args[1] instanceof Throwable) {
                        warnCount++;
                        lastMessage = (String) args[0];
                        lastThrowable = (Throwable) args[1];
                        return null;
                    }
                    // warn(String msg)
                    if (args.length == 1 && args[0] instanceof String) {
                        warnCount++;
                        lastMessage = (String) args[0];
                        lastThrowable = null;
                        return null;
                    }
                    // warn(String format, Object... args)
                    if (args.length >= 1 && args[0] instanceof String) {
                        warnCount++;
                        lastMessage = (String) args[0];
                        // try to find throwable in varargs if last arg is throwable
                        Object lastArg = args[args.length - 1];
                        if (lastArg instanceof Throwable) {
                            lastThrowable = (Throwable) lastArg;
                        } else {
                            lastThrowable = null;
                        }
                        return null;
                    }
                }
            }
            // For other methods, return sensible defaults
            Class<?> returnType = method.getReturnType();
            if (returnType == boolean.class) {
                return false;
            }
            if (returnType == int.class) {
                return 0;
            }
            return null;
        }

        int getWarnCount() {
            return warnCount;
        }

        String getLastMessage() {
            return lastMessage;
        }

        Throwable getLastThrowable() {
            return lastThrowable;
        }
    }
}
