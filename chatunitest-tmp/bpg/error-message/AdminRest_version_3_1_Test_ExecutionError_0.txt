public class AdminRest_version_3_1_Test {

    @Test
    public void version_shouldReturn200AndWriteJsonAndNewline() throws Exception {
        AdminRest adminRest = new AdminRest(new AppConfig(), new SimpleMeterRegistry());
        Response resp = adminRest.version();
        assertNotNull(resp, "Response should not be null");
        assertEquals(200, resp.getStatus(), "Response status should be 200");
        Object entity = resp.getEntity();
        assertNotNull(entity, "Response entity should not be null");
        assertTrue(entity instanceof RestStreamingOutput, "Entity should be a RestStreamingOutput");
        RestStreamingOutput streamingOutput = (RestStreamingOutput) entity;
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        // invoke the streaming output to write to our byte array stream
        streamingOutput.write(baos);
        String result = baos.toString(StandardCharsets.UTF_8.name());
        assertNotNull(result, "Output should not be null");
        assertFalse(result.isEmpty(), "Output should not be empty");
        // should end with system newline because writeLine appends System.lineSeparator()
        assertTrue(result.endsWith(System.lineSeparator()), "Output should end with a newline");
        // Basic sanity: valid JSON object or empty JSON produced (we don't know VersionInfo shape)
        String trimmed = result.trim();
        assertTrue(trimmed.startsWith("{") || trimmed.startsWith("\"") || trimmed.equals("{}"), "Output should look like a JSON object or be empty JSON");
    }

    @Test
    public void version_shouldHandleOutputStreamWriteFailuresGracefully() throws Exception {
        AdminRest adminRest = new AdminRest(new AppConfig(), new SimpleMeterRegistry());
        Response resp = adminRest.version();
        assertEquals(200, resp.getStatus());
        Object entity = resp.getEntity();
        assertTrue(entity instanceof RestStreamingOutput);
        RestStreamingOutput streamingOutput = (RestStreamingOutput) entity;
        // Create an OutputStream that throws on write to trigger the catch block in write()
        OutputStream failingOutput = new OutputStream() {

            @Override
            public void write(int b) throws IOException {
                throw new IOException("simulated write failure");
            }
        };
        // The implementation catches Throwable internally and should not propagate an exception.
        // If an exception is thrown here the test will fail.
        streamingOutput.write(failingOutput);
    }
}
