    // Helper to set private fields via reflection
    private static void setField(Object target, String fieldName, Object value) throws Exception {
        Field f = null;
        Class<?> cls = target.getClass();
        while (cls != null) {
            try {
                f = cls.getDeclaredField(fieldName);
                break;
            } catch (NoSuchFieldException e) {
                cls = cls.getSuperclass();
            }
        }
        if (f == null) {
            throw new NoSuchFieldException(fieldName);
        }
        f.setAccessible(true);
        f.set(target, value);
    }

    // Helper to invoke write(OutputStream) on the anonymous RestStreamingOutput entity
    private static void invokeWrite(Object entity, ByteArrayOutputStream out) throws NoSuchMethodException, IllegalAccessException, InvocationTargetException {
        Method write = entity.getClass().getMethod("write", java.io.OutputStream.class);
        write.invoke(entity, out);
    }

    @Test
    public void testGetLog_s3ObjectStreaming_extractsLogField() throws Exception {
        AppConfig appConfig = new AppConfig();
        appConfig.setSparkLogS3Bucket("my-bucket");
        SimpleMeterRegistry meterRegistry = new SimpleMeterRegistry();
        ApplicationGetLogRest rest = new ApplicationGetLogRest(appConfig, meterRegistry);
        User user = new User("charlie");
        // Prepare a stub LogDao that returns a single object key
        LogDao stubLogDao = new LogDao(null, null, null, null, meterRegistry) {

            @Override
            public ArrayList<String> getKeyList(String podPrefix, String execId) {
                ArrayList<String> keys = new ArrayList<>();
                keys.add("logs/part-000");
                return keys;
            }
        };
        // Prepare an S3Object that contains a JSON line with an escaped newline (\n) inside the log field.
        // The code's regex expects a literal backslash-n sequence, so we embed \\n in the content.
        String s3Line = "{\"log\":\"HELLO_WORLD\\\\n\",\"stream\":\"stdout\"}\n";
        byte[] bytes = s3Line.getBytes(StandardCharsets.UTF_8);
        S3Object s3Object = new S3Object();
        InputStream contentStream = new ByteArrayInputStream(bytes);
        s3Object.setObjectContent(contentStream);
        // Create a dynamic proxy for AmazonS3 that only handles getObject and returns our S3Object.
        Object amazonS3Proxy = Proxy.newProxyInstance(ApplicationGetLogRest.class.getClassLoader(), new Class[] { com.amazonaws.services.s3.AmazonS3.class }, (proxy, method, args) -> {
            if ("getObject".equals(method.getName()) && args != null && args.length >= 1 && args[0] instanceof GetObjectRequest) {
                return s3Object;
            }
            // Default: return null or throw for unexpected methods
            return null;
        });
        // Inject stubLogDao and s3Client into the rest instance
        setField(rest, "logDao", stubLogDao);
        setField(rest, "s3Client", amazonS3Proxy);
        // Use a valid 32-char submissionId (all 'a's)
        String submissionId = "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa";
        Response response = rest.getLog(submissionId, "", "true", "", "none", user);
        Object entity = response.getEntity();
        Assertions.assertNotNull(entity);
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        invokeWrite(entity, out);
        String output = out.toString(StandardCharsets.UTF_8.name());
        // The streaming output should contain the extracted HELLO_WORLD and at least one newline
        Assertions.assertTrue(output.contains("HELLO_WORLD"));
        Assertions.assertTrue(output.contains("\n"));
    }
