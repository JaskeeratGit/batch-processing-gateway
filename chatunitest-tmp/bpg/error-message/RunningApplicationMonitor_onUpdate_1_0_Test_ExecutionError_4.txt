/**
 * Unit tests for RunningApplicationMonitor.onUpdate(...)
 *
 * These tests avoid using Mockito to prevent MockMaker initialization issues in some test environments.
 * A simple anonymous implementation of AppConfig.SparkCluster is used instead.
 */
@ExtendWith(MockitoExtension.class)
public class RunningApplicationMonitor_onUpdate_1_0_Test {

    private Timer timer;

    private SimpleMeterRegistry meterRegistry;

    private AppConfig.SparkCluster sparkClusterMock;

    private RunningApplicationMonitor monitor;

    @BeforeAll
    public static void ensureMockitoInlineMockMaker() throws Exception {
        /*
         * Some environments fail to initialize Mockito's MockMaker plugin.
         * Create a mockito-extensions/org.mockito.plugins.MockMaker file in the test-classes directory
         * with content "mock-maker-inline" so Mockito can initialize the inline mock maker.
         */
        try {
            Path codeLocation = Paths.get(RunningApplicationMonitor_onUpdate_1_0_Test.class.getProtectionDomain().getCodeSource().getLocation().toURI());
            Path baseDir = codeLocation;
            // If codeLocation points to a jar file, fallback to target/test-classes in working dir
            if (Files.isRegularFile(codeLocation)) {
                baseDir = Paths.get(System.getProperty("user.dir"), "target", "test-classes");
            }
            if (!Files.exists(baseDir)) {
                // fallback: attempt to use current working directory target/test-classes
                baseDir = Paths.get(System.getProperty("user.dir"), "target", "test-classes");
            }
            Path extDir = baseDir.resolve("mockito-extensions");
            Files.createDirectories(extDir);
            Path pluginFile = extDir.resolve("org.mockito.plugins.MockMaker");
            // Write the inline mock maker identifier
            Files.write(pluginFile, "mock-maker-inline".getBytes(StandardCharsets.UTF_8));
        } catch (Exception e) {
            // If we fail to create the file, continue — tests may still work depending on environment.
            // Do not rethrow to avoid failing the test suite setup.
        }
    }

    @BeforeEach
    public void setup() {
        timer = new Timer(true);
        meterRegistry = new SimpleMeterRegistry();
        // Provide a simple anonymous implementation instead of using Mockito to avoid MockMaker issues.
        sparkClusterMock = new AppConfig.SparkCluster() {

            @Override
            public String getEksCluster() {
                return "";
            }

            @Override
            public String getSparkApplicationNamespace() {
                return "";
            }
        };
        // Use a large delete interval to avoid TimerTask triggering during tests
        monitor = new RunningApplicationMonitor(sparkClusterMock, timer, 60 * 60 * 1000L, meterRegistry);
    }

    @AfterEach
    public void teardown() {
        timer.cancel();
    }

    private ConcurrentHashMap<?, ?> getRunningApplicationsMap() throws Exception {
        Field f = RunningApplicationMonitor.class.getDeclaredField("runningApplications");
        f.setAccessible(true);
        @SuppressWarnings("unchecked")
        ConcurrentHashMap<?, ?> map = (ConcurrentHashMap<?, ?>) f.get(monitor);
        return map;
    }

    @Test
    public void testOnUpdate_stateNull_doesNothing() throws Exception {
        // Pass null as current CRD state; helper should handle null and nothing should be added.
        monitor.onUpdate(null, null);
        ConcurrentHashMap<?, ?> map = getRunningApplicationsMap();
        assertEquals(0, map.size());
    }

    @Test
    public void testOnUpdate_nameNull_logsAndDoesNothing() throws Exception {
        // Create a lightweight SparkApplication instance with metadata having null name.
        SparkApplication curr = instantiateSparkApplication();
        ObjectMeta meta = new ObjectMeta();
        meta.setName(null);
        setMetadata(curr, meta);
        // To make SparkApplicationResourceHelper.getState(...) return "RUNNING", we mock the static method.
        try (MockedStatic<SparkApplicationResourceHelper> mocked = mockStatic(SparkApplicationResourceHelper.class)) {
            mocked.when(() -> SparkApplicationResourceHelper.getState(curr)).thenReturn("RUNNING");
            monitor.onUpdate(null, curr);
        }
        ConcurrentHashMap<?, ?> map = getRunningApplicationsMap();
        assertEquals(0, map.size());
    }

    @Test
    public void testOnUpdate_addsRunningApplication_withAndWithoutLabel() throws Exception {
        // First: no labels -> should still add with default max running millis
        SparkApplication currNoLabel = instantiateSparkApplication();
        ObjectMeta metaNoLabel = new ObjectMeta();
        metaNoLabel.setName("app-no-label");
        metaNoLabel.setCreationTimestamp(Instant.now().toString());
        metaNoLabel.setNamespace("ns1");
        metaNoLabel.setLabels(null);
        setMetadata(currNoLabel, metaNoLabel);
        try (MockedStatic<SparkApplicationResourceHelper> mocked = mockStatic(SparkApplicationResourceHelper.class)) {
            mocked.when(() -> SparkApplicationResourceHelper.getState(currNoLabel)).thenReturn("RUNNING");
            monitor.onUpdate(null, currNoLabel);
        }
        ConcurrentHashMap<?, ?> map = getRunningApplicationsMap();
        assertEquals(1, map.size());
        // Second: provide label with a specific max running millis value
        SparkApplication currWithLabel = instantiateSparkApplication();
        ObjectMeta metaWithLabel = new ObjectMeta();
        metaWithLabel.setName("app-with-label");
        metaWithLabel.setCreationTimestamp(Instant.now().toString());
        metaWithLabel.setNamespace("ns2");
        Map<String, String> labels = Collections.singletonMap(Constants.MAX_RUNNING_MILLIS_LABEL, "12345");
        metaWithLabel.setLabels(labels);
        setMetadata(currWithLabel, metaWithLabel);
        try (MockedStatic<SparkApplicationResourceHelper> mocked = mockStatic(SparkApplicationResourceHelper.class)) {
            mocked.when(() -> SparkApplicationResourceHelper.getState(currWithLabel)).thenReturn("RUNNING");
            monitor.onUpdate(null, currWithLabel);
        }
        ConcurrentHashMap<?, ?> mapAfter = getRunningApplicationsMap();
        // now should have two entries (one from previous, one just added)
        assertEquals(2, mapAfter.size());
        // ensure keys include the expected namespace/name combinations by iterating keys and checking their toString contains expected values
        boolean foundNoLabel = false;
        boolean foundWithLabel = false;
        for (Object key : mapAfter.keySet()) {
            String s = key.toString();
            if (s.contains("ns1") && s.contains("app-no-label")) {
                foundNoLabel = true;
            }
            if (s.contains("ns2") && s.contains("app-with-label")) {
                foundWithLabel = true;
            }
        }
        assertTrue(foundNoLabel);
        assertTrue(foundWithLabel);
    }

    // Helper to instantiate SparkApplication reflectively to avoid Mockito usage for creation
    private SparkApplication instantiateSparkApplication() throws Exception {
        try {
            Class<?> clazz = Class.forName("com.apple.spark.operator.SparkApplication");
            Object inst = null;
            try {
                inst = clazz.getDeclaredConstructor().newInstance();
            } catch (NoSuchMethodException e) {
                // try without constructor
                inst = java.lang.reflect.Array.newInstance(clazz, 0);
                throw new IllegalStateException("Cannot instantiate SparkApplication", e);
            }
            return (SparkApplication) inst;
        } catch (ClassNotFoundException cnfe) {
            // If class not found, fall back to Mockito mock for the type (after ensuring inline mock-maker)
            @SuppressWarnings("unchecked")
            SparkApplication mock = (SparkApplication) Mockito.mock(SparkApplication.class);
            return mock;
        }
    }

    // Helper to set metadata on SparkApplication via reflection (handles both setter and field)
    private void setMetadata(SparkApplication app, ObjectMeta meta) throws Exception {
        if (app == null)
            return;
        try {
            java.lang.reflect.Method setter = app.getClass().getMethod("setMetadata", ObjectMeta.class);
            setter.invoke(app, meta);
            return;
        } catch (NoSuchMethodException e) {
            // try field
            try {
                Field f = app.getClass().getDeclaredField("metadata");
                f.setAccessible(true);
                f.set(app, meta);
                return;
            } catch (NoSuchFieldException ex) {
                // if neither present, and app is a Mockito mock, stub getMetadata()
                if (Mockito.mockingDetails(app).isMock()) {
                    when(app.getMetadata()).thenReturn(meta);
                    return;
                }
                throw new IllegalStateException("Unable to set metadata on SparkApplication", ex);
            }
        }
    }
}
