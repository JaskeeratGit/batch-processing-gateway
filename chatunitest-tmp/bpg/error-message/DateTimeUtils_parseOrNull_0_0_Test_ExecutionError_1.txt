    // Reflection helpers using VarHandle (works on Java 9+)
    private Logger getLoggerField() throws Exception {
        VarHandle vh = MethodHandles.privateLookupIn(DateTimeUtils.class, MethodHandles.lookup()).findStaticVarHandle(DateTimeUtils.class, "logger", Logger.class);
        return (Logger) vh.get();
    }

    private void setLoggerField(Logger newLogger) throws Exception {
        VarHandle vh = MethodHandles.privateLookupIn(DateTimeUtils.class, MethodHandles.lookup()).findStaticVarHandle(DateTimeUtils.class, "logger", Logger.class);
        vh.set(newLogger);
    }

    @Test
    void testParseOrNull_withInvalid_logsDebugAndReturnsNull() throws Exception {
        // save original logger
        originalLogger = getLoggerField();
        TestLogger testLogger = new TestLogger();
        setLoggerField(testLogger);
        String bad = "not-a-timestamp";
        Long result = DateTimeUtils.parseOrNull(bad);
        assertNull(result, "Should return null for invalid datetime");
        assertTrue(testLogger.debugCalled, "Debug should be called on parse failure");
        assertNotNull(testLogger.lastMessage, "Debug message should be present");
        assertTrue(testLogger.lastMessage.contains("Failed to parse datetime"));
        assertNotNull(testLogger.lastThrowable, "Throwable should be passed to debug");
    }

    @Test
    void testParseOrNull_withValid_returnsEpochMillis() {
        String iso = "2020-01-01T00:00:00Z";
        Long result = DateTimeUtils.parseOrNull(iso);
        assertEquals(Instant.parse(iso).toEpochMilli(), result);
    }

    @Test
    void testParseOrNull_withEmptyOrNull_returnsNullAndNoLogging() throws Exception {
        // save original logger
        originalLogger = getLoggerField();
        TestLogger testLogger = new TestLogger();
        setLoggerField(testLogger);
        assertNull(DateTimeUtils.parseOrNull(""), "Empty string should return null");
        assertNull(DateTimeUtils.parseOrNull(null), "Null should return null");
        // logger should not be invoked for empty or null (only for parse exceptions)
        assertFalse(testLogger.debugCalled, "Debug should not be called for null/empty input");
    }

    @Test
    void testParseOrNull_withNull_returnsNull() {
        assertNull(DateTimeUtils.parseOrNull(null));
    }

    @Test
    void testParseOrNull_withEmpty_returnsNull() {
        assertNull(DateTimeUtils.parseOrNull(""));
    }

    @Test
    void testParseOrNull_withValidIsoInstant_returnsEpochMillis() {
        // 2020-01-01T00:00:00Z => 1577836800000L
        String instant = "2020-01-01T00:00:00Z";
        Long result = DateTimeUtils.parseOrNull(instant);
        assertNotNull(result);
        assertEquals(1577836800000L, result.longValue());
    }
