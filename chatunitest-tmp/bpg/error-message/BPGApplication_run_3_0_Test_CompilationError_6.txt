@ExtendWith(MockitoExtension.class)
public class BPGApplication_run_3_0_Test {

    private Environment environment;

    private JerseyEnvironment jerseyEnvironment;

    private MutableServletContextHandler servletContextHandler;

    private com.codahale.metrics.health.HealthCheckRegistry healthCheckRegistry;

    private MetricRegistry metricRegistry;

    @BeforeEach
    public void setUp() {
        environment = mock(Environment.class);
        jerseyEnvironment = mock(JerseyEnvironment.class);
        servletContextHandler = mock(MutableServletContextHandler.class);
        healthCheckRegistry = mock(com.codahale.metrics.health.HealthCheckRegistry.class);
        metricRegistry = new MetricRegistry();
        when(environment.jersey()).thenReturn(jerseyEnvironment);
        when(environment.getApplicationContext()).thenReturn(servletContextHandler);
        when(environment.metrics()).thenReturn(metricRegistry);
        when(environment.healthChecks()).thenReturn(healthCheckRegistry);
        doNothing().when(jerseyEnvironment).register(any());
        doNothing().when(healthCheckRegistry).register(anyString(), any());
        doNothing().when(servletContextHandler).setContextPath(anyString());
    }

    @Test
    public void testRun_nullContextPath_usesDefaultAndRegistersResources() {
        AppConfig config = mock(AppConfig.class);
        when(config.getApplicationContextPath()).thenReturn(null);
        when(config.getAllowedUsers()).thenReturn(Collections.emptySet());
        when(config.getBlockedUsers()).thenReturn(Collections.emptySet());
        when(config.getSparkClusters()).thenReturn(Collections.emptyList());
        BPGApplication app = new BPGApplication(false);
        // Execute
        app.run(config, environment);
        // Verify default context path used
        verify(servletContextHandler).setContextPath(Constants.DEFAULT_APPLICATION_CONTEXT_PATH);
        // Verify that core REST resources are registered
        verify(jerseyEnvironment).register(isA(ApplicationSubmissionRest.class));
        verify(jerseyEnvironment).register(isA(ApplicationGetLogRest.class));
        verify(jerseyEnvironment).register(isA(CloudStorageRest.class));
        verify(jerseyEnvironment).register(isA(AdminRest.class));
        verify(jerseyEnvironment).register(isA(HealthcheckRest.class));
        // Verify exception mapper and auth bindings registered (at least once)
        verify(jerseyEnvironment, atLeastOnce()).register(isA(ThrowableExceptionMapper.class));
        verify(jerseyEnvironment, atLeastOnce()).register(isA(AuthDynamicFeature.class));
        verify(jerseyEnvironment, atLeastOnce()).register(isA(AuthValueFactoryProvider.Binder.class));
        // Verify health check registered
        verify(healthCheckRegistry).register(eq("sparkClusters"), isA(BPGHealthCheck.class));
        // Verify metric registry was added to SharedMetricRegistries
        assertSame(metricRegistry, SharedMetricRegistries.getOrCreate(Constants.DEFAULT_METRIC_REGISTRY));
    }

    @Test
    public void testRun_customContextPath_monitorApplicationTrue_and_invokePrivateSendPeriodicMetrics() throws Exception {
        AppConfig config = mock(AppConfig.class);
        when(config.getApplicationContextPath()).thenReturn("/custom");
        when(config.getAllowedUsers()).thenReturn(Collections.emptySet());
        when(config.getBlockedUsers()).thenReturn(Collections.emptySet());
        when(config.getSparkClusters()).thenReturn(Collections.emptyList());
        BPGApplication app = new BPGApplication(true);
        // Execute run (this will attempt to start ApplicationMonitor if available)
        app.run(config, environment);
        // Verify custom context path used
        verify(servletContextHandler).setContextPath("/custom");
        // Now invoke private method sendPeriodicMetrics reflectively
        Method m = BPGApplication.class.getDeclaredMethod("sendPeriodicMetrics", AppConfig.class, MeterRegistry.class);
        m.setAccessible(true);
        // Provide a simple MeterRegistry instance
        SimpleMeterRegistry simpleMeterRegistry = new SimpleMeterRegistry();
        // Should not throw
        m.invoke(app, config, simpleMeterRegistry);
    }
}
