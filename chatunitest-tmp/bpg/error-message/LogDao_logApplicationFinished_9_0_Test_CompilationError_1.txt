    private static void setField(Object target, String fieldName, Object value) throws Exception {
        Field f = findField(target.getClass(), fieldName);
        f.setAccessible(true);
        f.set(target, value);
    }

    private static Field findField(Class<?> clazz, String name) throws NoSuchFieldException {
        Class<?> c = clazz;
        while (c != null) {
            try {
                return c.getDeclaredField(name);
            } catch (NoSuchFieldException e) {
                c = c.getSuperclass();
            }
        }
        throw new NoSuchFieldException(name);
    }

    // A simple proxy-based PreparedStatement spy that records parameters and execute calls.
    private static class SpyPreparedStatementHandler implements InvocationHandler {

        final Map<Integer, Object> params = new HashMap<>();

        boolean executeCalled = false;

        final boolean throwOnExecute;

        SpyPreparedStatementHandler(boolean throwOnExecute) {
            this.throwOnExecute = throwOnExecute;
        }

        @Override
        public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
            String name = method.getName();
            if ("setString".equals(name) && args.length == 2) {
                params.put((Integer) args[0], args[1]);
                return null;
            }
            if ("setTimestamp".equals(name) && args.length == 2) {
                params.put((Integer) args[0], args[1]);
                return null;
            }
            if ("setDouble".equals(name) && args.length == 2) {
                params.put((Integer) args[0], args[1]);
                return null;
            }
            if ("executeUpdate".equals(name) && (args == null || args.length == 0)) {
                executeCalled = true;
                if (throwOnExecute) {
                    throw new SQLException("boom");
                }
                return 1;
            }
            // For close or other methods, just no-op or return default
            if ("close".equals(name)) {
                return null;
            }
            // default stub return
            if (method.getReturnType().isPrimitive()) {
                if (method.getReturnType() == boolean.class)
                    return false;
                if (method.getReturnType() == byte.class)
                    return (byte) 0;
                if (method.getReturnType() == short.class)
                    return (short) 0;
                if (method.getReturnType() == int.class)
                    return 0;
                if (method.getReturnType() == long.class)
                    return 0L;
                if (method.getReturnType() == float.class)
                    return 0f;
                if (method.getReturnType() == double.class)
                    return 0d;
                if (method.getReturnType() == char.class)
                    return '\0';
            }
            return null;
        }
    }

    // Connection proxy that returns the preparedStatement proxy for prepareStatement calls.
    private static Connection createConnectionProxy(PreparedStatement psProxy) {
        InvocationHandler handler = (proxy, method, args) -> {
            String name = method.getName();
            if ("prepareStatement".equals(name) && args != null && args.length >= 1) {
                return psProxy;
            }
            if ("close".equals(name)) {
                return null;
            }
            // default stub behavior
            if (method.getReturnType().isPrimitive()) {
                if (method.getReturnType() == boolean.class)
                    return false;
                if (method.getReturnType() == byte.class)
                    return (byte) 0;
                if (method.getReturnType() == short.class)
                    return (short) 0;
                if (method.getReturnType() == int.class)
                    return 0;
                if (method.getReturnType() == long.class)
                    return 0L;
                if (method.getReturnType() == float.class)
                    return 0f;
                if (method.getReturnType() == double.class)
                    return 0d;
                if (method.getReturnType() == char.class)
                    return '\0';
            }
            return null;
        };
        return (Connection) Proxy.newProxyInstance(Connection.class.getClassLoader(), new Class[] { Connection.class }, handler);
    }

    @Test
    public void testLogApplicationFinishedHandlesExceptionAndIncrementsFailureMetric() throws Exception {
        LogDao dao = new LogDao("", "u", "p", "mydb", new LoggingMeterRegistry());
        // Prepare a spy PreparedStatement that throws SQLException on executeUpdate
        SpyPreparedStatementHandler psHandler = new SpyPreparedStatementHandler(true);
        PreparedStatement psProxy = (PreparedStatement) Proxy.newProxyInstance(PreparedStatement.class.getClassLoader(), new Class[] { PreparedStatement.class }, psHandler);
        Connection connProxy = createConnectionProxy(psProxy);
        DBConnection dbConnStub = new DBConnection() {

            @Override
            public Connection getConnection() throws SQLException {
                return connProxy;
            }
        };
        setField(dao, "dbConnection", dbConnStub);
        setField(dao, "bypassLog", false);
        TimerMetricContainer immediateTimer = new TimerMetricContainer(new LoggingMeterRegistry()) {

            @Override
            public void record(Runnable runnable, String metricName, io.micrometer.core.instrument.Tag... tags) {
                runnable.run();
            }
        };
        setField(dao, "timerMetrics", immediateTimer);
        // Spy failure metrics to capture increment call
        class SpyFailureMetrics extends CounterMetricContainer {

            boolean incremented = false;

            String metricName;

            io.micrometer.core.instrument.Tag[] tags;

            SpyFailureMetrics() {
                super(new LoggingMeterRegistry());
            }

            @Override
            public void increment(String metricName, io.micrometer.core.instrument.Tag... tags) {
                incremented = true;
                this.metricName = metricName;
                this.tags = tags;
            }
        }
        SpyFailureMetrics spyFailure = new SpyFailureMetrics();
        setField(dao, "failureMetrics", spyFailure);
        // Invoke - should not throw despite underlying SQLException being thrown, since Dao likely catches it
        dao.logApplicationFinished("sub-err", "FAILED", new Timestamp(System.currentTimeMillis()), 0.5, 0.5);
        // Verify executeUpdate attempted
        assertTrue(psHandler.executeCalled, "executeUpdate should have been attempted");
        // Verify failureMetrics.increment was called
        assertTrue(spyFailure.incremented, "failure metric should have been incremented on exception");
        assertNotNull(spyFailure.metricName);
        assertNotNull(spyFailure.tags);
        boolean foundOperation = false;
        boolean foundException = false;
        for (Tag t : spyFailure.tags) {
            if ("operation".equals(t.getKey()) && "log_app_finished".equals(t.getValue())) {
                foundOperation = true;
            }
            if ("exception".equals(t.getKey()) && "SQLException".equals(t.getValue())) {
                foundException = true;
            }
        }
        assertTrue(foundOperation, "operation tag with log_app_finished should be present");
        assertTrue(foundException, "exception tag with SQLException should be present");
    }
