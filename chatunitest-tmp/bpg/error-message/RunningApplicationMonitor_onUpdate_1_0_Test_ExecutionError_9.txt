/**
 * Unit tests for RunningApplicationMonitor.onUpdate(...)
 *
 * Note: This test file provides a lightweight test-only implementation of
 * SparkApplicationResourceHelper (same package & class name) to control
 * the returned state without requiring Mockito static mocking.
 */
public class RunningApplicationMonitor_onUpdate_1_0_Test {

    private Timer timer;

    private SimpleMeterRegistry meterRegistry;

    private AppConfig.SparkCluster sparkClusterMock;

    private RunningApplicationMonitor monitor;

    @BeforeEach
    public void setup() {
        timer = new Timer(true);
        meterRegistry = new SimpleMeterRegistry();
        // Provide a simple anonymous implementation instead of using Mockito.
        sparkClusterMock = new AppConfig.SparkCluster() {

            @Override
            public String getEksCluster() {
                return "";
            }

            @Override
            public String getSparkApplicationNamespace() {
                return "";
            }
        };
        // Use a large delete interval to avoid TimerTask triggering during tests
        monitor = new RunningApplicationMonitor(sparkClusterMock, timer, 60 * 60 * 1000L, meterRegistry);
    }

    @AfterEach
    public void teardown() {
        timer.cancel();
        // clear test helper state
        SparkApplicationResourceHelper.clearForTests();
    }

    private ConcurrentHashMap<?, ?> getRunningApplicationsMap() throws Exception {
        Field f = RunningApplicationMonitor.class.getDeclaredField("runningApplications");
        f.setAccessible(true);
        @SuppressWarnings("unchecked")
        ConcurrentHashMap<?, ?> map = (ConcurrentHashMap<?, ?>) f.get(monitor);
        return map;
    }

    @Test
    public void testOnUpdate_stateNull_doesNothing() throws Exception {
        // Helper returns NOT_RUNNING by default for null input
        monitor.onUpdate(null, null);
        ConcurrentHashMap<?, ?> map = getRunningApplicationsMap();
        assertEquals(0, map.size());
    }

    @Test
    public void testOnUpdate_nameNull_logsAndDoesNothing() throws Exception {
        SparkApplication curr = instantiateSparkApplication();
        ObjectMeta meta = new ObjectMeta();
        meta.setName(null);
        setMetadata(curr, meta);
        SparkApplicationResourceHelper.setStateForTests(curr, "RUNNING");
        monitor.onUpdate(null, curr);
        ConcurrentHashMap<?, ?> map = getRunningApplicationsMap();
        assertEquals(0, map.size());
    }

    @Test
    public void testOnUpdate_creationTimestampNull_doesNothing() throws Exception {
        SparkApplication curr = instantiateSparkApplication();
        ObjectMeta meta = new ObjectMeta();
        meta.setName("app1");
        meta.setCreationTimestamp(null);
        setMetadata(curr, meta);
        SparkApplicationResourceHelper.setStateForTests(curr, "RUNNING");
        monitor.onUpdate(null, curr);
        ConcurrentHashMap<?, ?> map = getRunningApplicationsMap();
        assertEquals(0, map.size());
    }

    @Test
    public void testOnUpdate_invalidCreationTimestamp_doesNothing() throws Exception {
        SparkApplication curr = instantiateSparkApplication();
        ObjectMeta meta = new ObjectMeta();
        meta.setName("app2");
        meta.setCreationTimestamp("not-a-timestamp");
        setMetadata(curr, meta);
        SparkApplicationResourceHelper.setStateForTests(curr, "RUNNING");
        monitor.onUpdate(null, curr);
        ConcurrentHashMap<?, ?> map = getRunningApplicationsMap();
        assertEquals(0, map.size());
    }

    @Test
    public void testOnUpdate_namespaceNull_doesNothing() throws Exception {
        SparkApplication curr = instantiateSparkApplication();
        ObjectMeta meta = new ObjectMeta();
        meta.setName("app3");
        meta.setCreationTimestamp(Instant.now().toString());
        meta.setNamespace(null);
        setMetadata(curr, meta);
        SparkApplicationResourceHelper.setStateForTests(curr, "RUNNING");
        monitor.onUpdate(null, curr);
        ConcurrentHashMap<?, ?> map = getRunningApplicationsMap();
        assertEquals(0, map.size());
    }

    @Test
    public void testOnUpdate_addsRunningApplication_withAndWithoutLabel() throws Exception {
        // First: no labels -> should still add with default max running millis
        SparkApplication currNoLabel = instantiateSparkApplication();
        ObjectMeta metaNoLabel = new ObjectMeta();
        metaNoLabel.setName("app-no-label");
        metaNoLabel.setCreationTimestamp(Instant.now().toString());
        metaNoLabel.setNamespace("ns1");
        metaNoLabel.setLabels(null);
        setMetadata(currNoLabel, metaNoLabel);
        SparkApplicationResourceHelper.setStateForTests(currNoLabel, "RUNNING");
        monitor.onUpdate(null, currNoLabel);
        ConcurrentHashMap<?, ?> map = getRunningApplicationsMap();
        assertEquals(1, map.size());
        // Second: provide label with a specific max running millis value
        SparkApplication currWithLabel = instantiateSparkApplication();
        ObjectMeta metaWithLabel = new ObjectMeta();
        metaWithLabel.setName("app-with-label");
        metaWithLabel.setCreationTimestamp(Instant.now().toString());
        metaWithLabel.setNamespace("ns2");
        Map<String, String> labels = Collections.singletonMap(Constants.MAX_RUNNING_MILLIS_LABEL, "12345");
        metaWithLabel.setLabels(labels);
        setMetadata(currWithLabel, metaWithLabel);
        SparkApplicationResourceHelper.setStateForTests(currWithLabel, "RUNNING");
        monitor.onUpdate(null, currWithLabel);
        ConcurrentHashMap<?, ?> mapAfter = getRunningApplicationsMap();
        // now should have two entries (one from previous, one just added)
        assertEquals(2, mapAfter.size());
        // ensure keys include the expected namespace/name combinations by iterating keys and checking their toString contains expected values
        boolean foundNoLabel = false;
        boolean foundWithLabel = false;
        for (Object key : mapAfter.keySet()) {
            String s = key.toString();
            if (s.contains("ns1") && s.contains("app-no-label")) {
                foundNoLabel = true;
            }
            if (s.contains("ns2") && s.contains("app-with-label")) {
                foundWithLabel = true;
            }
        }
        assertTrue(foundNoLabel);
        assertTrue(foundWithLabel);
    }

    // Helper to instantiate SparkApplication (production class)
    private SparkApplication instantiateSparkApplication() {
        return new SparkApplication();
    }

    // Helper to set metadata on SparkApplication via setter or field
    private void setMetadata(SparkApplication app, ObjectMeta meta) throws Exception {
        if (app == null)
            return;
        try {
            java.lang.reflect.Method setter = app.getClass().getMethod("setMetadata", ObjectMeta.class);
            setter.invoke(app, meta);
            return;
        } catch (NoSuchMethodException e) {
            // try field
            try {
                Field f = app.getClass().getDeclaredField("metadata");
                f.setAccessible(true);
                f.set(app, meta);
                return;
            } catch (NoSuchFieldException ex) {
                throw new IllegalStateException("Unable to set metadata on SparkApplication", ex);
            }
        }
    }
}

/*
 A lightweight test-only replacement/shadow for the production
 com.apple.spark.core.SparkApplicationResourceHelper. This class is
 intentionally package-private and is compiled in the test sources to
 allow controlling the returned state without requiring Mockito static
 mocking (mockito-inline).
*/
class SparkApplicationResourceHelper {

    private static final java.util.Map<SparkApplication, String> stateMap = java.util.Collections.synchronizedMap(new java.util.IdentityHashMap<>());

    /**
     * Called by RunningApplicationMonitor in production code.
     * Returns a state for the given SparkApplication as configured by tests.
     * If no explicit state has been set for the given instance, default to "NOT_RUNNING".
     */
    public static String getState(SparkApplication app) {
        if (app == null) {
            return "NOT_RUNNING";
        }
        String s = stateMap.get(app);
        return s != null ? s : "NOT_RUNNING";
    }

    // Test helper: set desired state for a SparkApplication instance
    public static void setStateForTests(SparkApplication app, String state) {
        if (app == null) {
            return;
        }
        stateMap.put(app, state);
    }

    // Test helper: clear configured states
    public static void clearForTests() {
        stateMap.clear();
    }
}
